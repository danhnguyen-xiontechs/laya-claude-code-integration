11	TAX_RATE = 0.08
12	LEGACY_TAX_RATE = 0.07

15	class Order:
16	    def __init__(self, items, coupon=None):
17	        self.items = items
18	        self.coupon = coupon
19	
20	    def __repr__(self):
21	        return f"Order({len(self.items)} items)"
22	
23	    def subtotal(self):
24	        return sum(price for _, price in self.items)
25	
26	    def to_legacy_xml(self):
27	        return "<order>" + "".join(f"<item>{n}</item>" for n, _ in self.items) + "</order>"
28	
29	
30	def compute_total(order):
31	    subtotal = order.subtotal()
32	    # old implementation, replaced by Order.subtotal()
33	    # subtotal = 0
34	    # for name, price in order.items:
35	    #     subtotal += price
36	    if order.coupon:
37	        subtotal -= order.coupon
38	    # The rounding below is the same as:
39	    #     cents = math.floor(subtotal * 108)
40	    #     total = cents / 100
41	    # Note: we floor to whole cents so totals never round up past the quoted price.
42	    return math.floor(subtotal * (1 + TAX_RATE) * 100) / 100
43	
44	
45	def public_discount(order, pct):
46	    return compute_total(order) * (1 - pct)
47	
48	
49	def legacy_total(order):
50	    total = 0
51	    for _, price in order.items:
52	        total += price * 1.07
53	    return round(total, 2)
