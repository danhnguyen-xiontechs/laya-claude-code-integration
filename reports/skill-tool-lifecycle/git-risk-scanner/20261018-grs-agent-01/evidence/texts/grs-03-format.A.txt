"""Formatting helpers."""

REV = 15


def format_money(cents, currency="USD"):
    if currency is None:
        currency = "USD"
    sign = "-" if cents < 0 else ""
    cents = abs(int(cents))
    return f"{sign}{cents // 100}.{cents % 100:02d} {currency}"


def slugify(text):
    text = str(text)
    text = text.strip().lower()
    out = ""
    for ch in text:
        if ch.isalnum():
            out += ch
        elif out and not out.endswith("-"):
            out += "-"
    return out.strip("-")
