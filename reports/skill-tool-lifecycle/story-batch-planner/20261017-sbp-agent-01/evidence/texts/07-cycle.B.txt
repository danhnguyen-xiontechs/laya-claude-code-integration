4	import { spawnSync } from 'node:child_process';
5	import fs from 'node:fs';
6	import path from 'node:path';
7	const CASE = "07-cycle";
8	const PLANNER = "D:\\XionTechs\\Laya\\.claude\\skills\\story-batch-planner\\scripts\\plan.mjs";
9	const LOGS = "C:\\Users\\ADMIN\\AppData\\Local\\Temp\\claude\\D--XionTechs-Laya--claude-worktrees-skill-build-workflow-review-af9324\\82e30143-e152-4ee9-b306-ffbd1654c94d\\scratchpad\\sbp\\_logs";

15	const r = spawnSync(process.execPath, [PLANNER, '--in', path.join(here, 'batch.json'), '--out', planJson, '--jev', '--summary'], {
16	  encoding: 'utf8',
17	  env: { ...process.env, HOME, USERPROFILE: HOME },

22	const jev = plan.edges.filter((e) => e.jev_suggestion).map((e) => ({
23	  pair: e.pair, deterministic: e.relation, jev: e.jev_suggestion.label, p: e.jev_suggestion.p ?? null,
24	  valid: e.jev_suggestion.valid, error: e.jev_suggestion.error ?? null, latencyMs: e.jev_suggestion.provenance?.latencyMs ?? null,
25	}));
26	fs.appendFileSync(path.join(LOGS, CASE + '.B.jsonl'), JSON.stringify({
27	  t: new Date().toISOString(), case: CASE, arm: 'B', waves: plan.waves.map((w) => w.stories), jev_shadow: plan.jev_shadow ?? null, jev,
28	}) + '\n');
